// @vitest-environment jsdom
import {
  act,
  cleanup,
  render,
  screen,
  waitFor,
  within,
} from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest'

import { getMessage } from '@/features/i18n/lib/language'
import { OptionsRoute } from '@/features/options/routes/OptionsRoute'
import type { StorageChange } from '@/lib/browser/chrome-storage'
import { defaultSettings, getUserSettings } from '@/lib/storage/settings'

type ChangeListener = (
  changes: Record<string, StorageChange>,
  areaName: string,
) => void

const storage = vi.hoisted(() => ({
  values: {} as Record<string, unknown>,
  listeners: new Set<ChangeListener>(),
  get: vi.fn(async () => storage.values),
  set: vi.fn(async (values: Record<string, unknown>) => {
    Object.assign(storage.values, values)
  }),
}))

vi.mock('@/lib/browser/chrome-storage', () => ({
  getChromeStorageLocal: () => ({ get: storage.get, set: storage.set }),
  getChromeStorageOnChanged: () => ({
    addListener: (listener: ChangeListener) => storage.listeners.add(listener),
    removeListener: (listener: ChangeListener) =>
      storage.listeners.delete(listener),
  }),
  warnMissingChromeStorage: vi.fn(),
}))
vi.mock('@/features/i18n/context/I18nProvider', () => ({
  useI18n: () => ({ t: (key: string) => getMessage('en', key) }),
}))
vi.mock('@/components/ModeToggle', () => ({ ModeToggle: () => null }))
vi.mock('@/components/ui/sonner', () => ({ Toaster: () => null }))
vi.mock('@/features/i18n/components/LanguageSelect', () => ({
  LanguageSelect: () => null,
}))
vi.mock('@/features/options/ImportExportSettings', () => ({
  ImportExportSettings: () => null,
}))
vi.mock('@/features/options/hooks/useColorSettings', () => ({
  useColorSettings: () => ({
    handleColorChange: vi.fn(),
    handleResetColors: vi.fn(),
  }),
}))

const scrollIntoView = Object.getOwnPropertyDescriptor(
  HTMLElement.prototype,
  'scrollIntoView',
)

beforeEach(() => {
  vi.clearAllMocks()
  vi.spyOn(console, 'log').mockImplementation(() => undefined)
  storage.values = { userSettings: { ...defaultSettings, language: 'en' } }
  storage.listeners.clear()
  Object.defineProperty(HTMLElement.prototype, 'scrollIntoView', {
    configurable: true,
    value: vi.fn(),
  })
})
afterEach(() => {
  cleanup()
  vi.restoreAllMocks()
  if (scrollIntoView) {
    Object.defineProperty(
      HTMLElement.prototype,
      'scrollIntoView',
      scrollIntoView,
    )
  } else {
    Reflect.deleteProperty(HTMLElement.prototype, 'scrollIntoView')
  }
})

const renderOptions = async () => {
  render(<OptionsRoute />)
  // Wait for the existing page to load before asserting the new control.
  await screen.findByRole('heading', {
    name: getMessage('en', 'options.title'),
  })
  return screen.queryByRole('region', { name: 'Ollama connection' })
}

describe('Issue 671 options UI and storage integration', () => {
  it('offers exactly the allowed hosts, persists a change, and reads it after remount', async () => {
    const settings = { ...defaultSettings, language: 'en' as const }
    Reflect.set(settings, 'ollamaBaseUrl', 'http://127.0.0.1:11434')
    storage.values.userSettings = settings
    const region = await renderOptions()
    expect(region).not.toBeNull()
    if (!region) {
      return
    }
    const ui = within(region)
    const selector = ui.getByRole('combobox')
    expect(selector.textContent).toBe('http://127.0.0.1:11434')
    expect(ui.queryByRole('textbox')).toBeNull()
    const user = userEvent.setup()
    selector.focus()
    await user.keyboard('{Enter}')
    expect(
      screen.getAllByRole('option').map((option) => option.textContent),
    ).toEqual(['http://localhost:11434', 'http://127.0.0.1:11434'])
    await user.keyboard('{Home}{Enter}')
    await waitFor(async () => {
      expect(Reflect.get(await getUserSettings(), 'ollamaBaseUrl')).toBe(
        'http://localhost:11434',
      )
    })
    cleanup()
    const reloadedRegion = await renderOptions()
    expect(reloadedRegion).not.toBeNull()
    if (reloadedRegion) {
      expect(within(reloadedRegion).getByRole('combobox').textContent).toBe(
        'http://localhost:11434',
      )
    }
  })

  it.each([undefined, 'https://remote.example:11434'])(
    'shows the safe default for absent or corrupt stored host %s',
    async (host) => {
      storage.values.userSettings = { language: 'en', ollamaBaseUrl: host }
      const region = await renderOptions()
      expect(region).not.toBeNull()
      if (region) {
        expect(within(region).getByRole('combobox').textContent).toBe(
          'http://localhost:11434',
        )
        expect(within(region).queryByRole('textbox')).toBeNull()
      }
    },
  )

  it('resets a loaded IP selection when settings are removed in another context', async () => {
    const settings = { ...defaultSettings, language: 'en' as const }
    Reflect.set(settings, 'ollamaBaseUrl', 'http://127.0.0.1:11434')
    storage.values.userSettings = settings
    const region = await renderOptions()
    expect(region).not.toBeNull()
    if (!region) {
      return
    }
    expect(within(region).getByRole('combobox').textContent).toBe(
      'http://127.0.0.1:11434',
    )
    await act(async () => {
      delete storage.values.userSettings
      for (const listener of storage.listeners) {
        listener({ userSettings: { oldValue: settings } }, 'local')
      }
    })
    expect(within(region).getByRole('combobox').textContent).toBe(
      'http://localhost:11434',
    )
  })
})
